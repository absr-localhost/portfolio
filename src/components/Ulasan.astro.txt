<section id="ulasan" class={`${wrap} pb-14 sm:pb-16`}>
  <div class="mb-6 flex flex-wrap items-end justify-between gap-3">
    <div>
      <h2 class="text-2xl font-bold text-heading">Ulasan klien</h2>
      <p id="ul-ringkas" class="mt-1 text-sm text-muted"></p>
    </div>
    <button id="ul-buka" type="button" class="rounded-md bg-primary px-4 py-2 text-sm font-semibold text-white transition hover:bg-secondary">Tulis ulasan</button>
  </div>

  <div id="ul-daftar" class="grid gap-4 sm:grid-cols-2 lg:grid-cols-3"></div>
  <div class="mt-6 text-center">
    <button id="ul-lagi" type="button" hidden class="text-sm font-semibold text-link underline underline-offset-4"></button>
  </div>

  <dialog id="ul-dialog" class="m-auto w-[calc(100%-2rem)] max-w-md rounded-lg border border-slate-200 bg-white p-0 shadow-xl backdrop:bg-black/40">
    <form id="ul-form" class="p-5" novalidate>
      <h3 class="text-lg font-bold text-heading">Tulis ulasan</h3>
      <label class="mt-4 block text-sm font-semibold text-heading" for="ul-nama">Nama</label>
      <input id="ul-nama" name="nama" required minlength="2" maxlength="60" autocomplete="name" class="mt-1 w-full rounded-md border border-slate-300 px-3 py-2 text-sm text-default" />
      <label class="mt-3 block text-sm font-semibold text-heading" for="ul-peran">Instansi atau jabatan (boleh dikosongkan)</label>
      <input id="ul-peran" name="peran" maxlength="80" class="mt-1 w-full rounded-md border border-slate-300 px-3 py-2 text-sm text-default" />
      <label class="mt-3 block text-sm font-semibold text-heading" for="ul-nilai">Nilai</label>
      <select id="ul-nilai" name="bintang" class="mt-1 w-full rounded-md border border-slate-300 bg-white px-3 py-2 text-sm text-default">
        <option value="5">5 dari 5</option>
        <option value="4">4 dari 5</option>
        <option value="3">3 dari 5</option>
        <option value="2">2 dari 5</option>
        <option value="1">1 dari 5</option>
      </select>
      <label class="mt-3 block text-sm font-semibold text-heading" for="ul-teks">Ulasan</label>
      <textarea id="ul-teks" name="teks" rows="4" required minlength="15" maxlength="500" class="mt-1 w-full rounded-md border border-slate-300 px-3 py-2 text-sm text-default"></textarea>
      <input name="situs" tabindex="-1" autocomplete="off" aria-hidden="true" class="hidden" />
      <p id="ul-pesan" class="mt-3 min-h-5 text-sm text-muted" role="status"></p>
      <div class="mt-4 flex justify-end gap-3">
        <button id="ul-batal" type="button" class="rounded-md border border-slate-300 px-4 py-2 text-sm font-semibold text-heading">Tutup</button>
        <button id="ul-kirim" type="submit" class="rounded-md bg-primary px-4 py-2 text-sm font-semibold text-white transition hover:bg-secondary">Kirim</button>
      </div>
    </form>
  </dialog>
</section>

<script>
  const API = 'https://ulasan-klien.extreamcollectiveofficial.workers.dev/api/ulasan';
  const BATAS = 6;
  const daftar = document.getElementById('ul-daftar');
  const ringkas = document.getElementById('ul-ringkas');
  const lagi = document.getElementById('ul-lagi');
  const dialog = document.getElementById('ul-dialog');
  const form = document.getElementById('ul-form');
  const pesan = document.getElementById('ul-pesan');
  const kirim = document.getElementById('ul-kirim');

  if (daftar && ringkas && lagi && dialog && form && pesan && kirim) {
    let data = [];
    let avg = 0;
    let semua = false;

    const el = (tag, cls, text) => {
      const e = document.createElement(tag);
      if (cls) e.className = cls;
      if (text) e.textContent = text;
      return e;
    };

    const tgl = (s) => {
      const d = new Date(s + 'T00:00:00');
      return isNaN(d.getTime()) ? '' : d.toLocaleDateString('id-ID', { day: 'numeric', month: 'short', year: 'numeric' });
    };

    const render = () => {
      daftar.textContent = '';
      if (!data.length) {
        ringkas.textContent = '';
        lagi.hidden = true;
        return;
      }
      ringkas.textContent = String(avg).replace('.', ',') + ' dari 5 (' + data.length + ' ulasan)';
      data.slice(0, semua ? data.length : BATAS).forEach((r) => {
        const kartu = el('article', 'rounded-lg border border-slate-200 bg-white p-5 shadow-sm');
        kartu.append(el('p', 'font-bold text-heading', r.nama));
        if (r.peran) kartu.append(el('p', 'text-sm text-muted', r.peran));
        const baris = el('div', 'mt-2 flex items-center justify-between gap-3');
        const bt = el('span', 'text-sm tracking-widest text-heading', '★'.repeat(r.bintang) + '☆'.repeat(5 - r.bintang));
        bt.setAttribute('aria-label', r.bintang + ' dari 5');
        baris.append(bt, el('span', 'text-xs text-muted', tgl(r.tanggal)));
        kartu.append(baris, el('p', 'mt-3 text-sm leading-relaxed text-default', r.teks));
        daftar.append(kartu);
      });
      lagi.hidden = data.length <= BATAS;
      lagi.textContent = semua ? 'Tampilkan lebih sedikit' : 'Tampilkan semua ' + data.length + ' ulasan';
    };

    const muat = () =>
      fetch(API, { cache: 'no-store' })
        .then((r) => r.json())
        .then((j) => {
          data = Array.isArray(j.items) ? j.items : [];
          avg = j.avg || 0;
          render();
        })
        .catch(() => {});
    muat();

    lagi.addEventListener('click', () => {
      semua = !semua;
      render();
    });

    document.getElementById('ul-buka')?.addEventListener('click', () => {
      pesan.textContent = '';
      dialog.showModal();
    });
    document.getElementById('ul-batal')?.addEventListener('click', () => dialog.close());

    form.addEventListener('submit', async (e) => {
      e.preventDefault();
      const f = new FormData(form);
      const body = { nama: f.get('nama'), peran: f.get('peran'), bintang: f.get('bintang'), teks: f.get('teks'), situs: f.get('situs') };
      pesan.textContent = 'Mengirim...';
      kirim.disabled = true;
      try {
        const r = await fetch(API, { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify(body) });
        const j = await r.json().catch(() => ({}));
        if (r.ok) {
          pesan.textContent = 'Terima kasih, ulasan Anda sudah tampil.';
          form.reset();
          muat();
          setTimeout(() => dialog.close(), 1500);
        } else {
          pesan.textContent = j.error || 'Gagal mengirim, coba lagi.';
        }
      } catch {
        pesan.textContent = 'Gagal mengirim. Periksa koneksi, lalu coba lagi.';
      }
      kirim.disabled = false;
    });
  }
</script>
